cmake_minimum_required(VERSION 3.22)
project(hexagon_fluid_simulation VERSION 0.1.0 LANGUAGES CXX)
option(HFS_BUILD_BENCHMARKS "Build the standalone CPU benchmark" ON)
option(HFS_ENABLE_SANITIZERS "Enable ASan and UBSan on desktop GCC/Clang" OFF)

add_library(hfs_core STATIC
    native/core/src/poisson.cpp
    native/cpu/src/jacobi.cpp)
target_compile_features(hfs_core PUBLIC cxx_std_20)
target_include_directories(hfs_core PUBLIC
    native/core/include
    native/cpu/include)
if(MSVC)
    target_compile_options(hfs_core PRIVATE /W4)
else()
    target_compile_options(hfs_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ANDROID)
    set_target_properties(hfs_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(hfs_jni SHARED app/src/main/cpp/hfs_jni.cpp)
    target_compile_features(hfs_jni PRIVATE cxx_std_20)
    target_link_libraries(hfs_jni PRIVATE hfs_core)
endif()

if(HFS_ENABLE_SANITIZERS)
    if(ANDROID OR MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizers require desktop GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(HFS_BUILD_BENCHMARKS)
    add_executable(hfs_bench benchmarks/poisson_benchmark.cpp)
    target_link_libraries(hfs_bench PRIVATE hfs_core)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(hfs_tests tests/poisson_tests.cpp)
    target_link_libraries(hfs_tests PRIVATE hfs_core)
    add_test(NAME hfs_cpu_poisson COMMAND hfs_tests)
endif()
